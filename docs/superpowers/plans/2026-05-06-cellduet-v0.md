```python
# src/cellduet/inchikey.py
"""InChIKey computation from SMILES, used for cross-dataset compound joining."""
from typing import Literal, Optional

from rdkit import Chem


def compute_inchikey(
    smiles: str,
    key_type: Literal["full", "skeleton"] = "full",
) -> Optional[str]:
    """Compute the InChIKey for a SMILES string.

    Returns None for unparseable SMILES. Skeleton key is the first 14 chars
    (connectivity layer only), full key is 27 chars (connectivity + stereochemistry + protonation).
    """
    if key_type not in ("full", "skeleton"):
        raise ValueError(f"key_type must be 'full' or 'skeleton', got {key_type!r}")
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    full = Chem.MolToInchiKey(mol)
    return full if key_type == "full" else full[:14]
```

```python
# src/cellduet/tahoe.py
"""Tahoe-100M streaming loader and per-drug aggregation helpers.

The full Tahoe expression atlas is 337 GB on Hugging Face; this module
streams the `pseudobulk_differential_expression` config and aggregates
to per-(drug, cell_line) log-fold-change vectors. See
docs/datasets/tahoe-100m.md for distribution + license details.
"""
from typing import Iterable, List

import pandas as pd


def plate_match_dmso(df: pd.DataFrame, dmso_label: str = "DMSO_TF") -> pd.DataFrame:
    """Drop drug rows on plates that have no DMSO control row.

    Tahoe pairs (drug, plate) against (DMSO_TF, plate) for the LFC contrast.
    This function enforces that a drug row's plate must also carry DMSO_TF.
    """
    plates_with_dmso = set(df.loc[df["drug"] == dmso_label, "plate"].unique())
    return df[(df["drug"] != dmso_label) & (df["plate"].isin(plates_with_dmso))]


def aggregate_per_drug(
    df: pd.DataFrame,
    value_cols: Iterable[str],
    group_cols: List[str] = None,
) -> pd.DataFrame:
    """Mean-aggregate value_cols across replicate rows within each (drug, cell_line).

    Returns a dataframe indexed by (drug, cell_line) with one column per value_col.
    """
    if group_cols is None:
        group_cols = ["drug", "cell_line"]
    return df.groupby(group_cols)[list(value_cols)].mean()


def stream_pseudobulk_de_to_tall_df(
    config: str = "pseudobulk_differential_expression",
    repo_id: str = "tahoebio/Tahoe-100M",
    drugs_filter: Iterable[str] = None,
    n_genes: int = 2000,
):
    """Stream Tahoe's pre-computed pseudobulk DE table from HF, optionally filtered.

    Returns a tall pandas dataframe with columns at minimum:
    [sample, drug, cell_line, plate, gene_symbol, log_fold_change].
    Use HVG selection (n_genes) to keep memory bounded.

    NOTE: column names in the parquet are confirmed empirically in notebook 02
    before this function is called. Fix here if the schema differs.
    """
    from datasets import load_dataset

    ds = load_dataset(repo_id, config, streaming=True, split="train")
    rows = []
    for batch in ds.iter(batch_size=10_000):
        df = pd.DataFrame(batch)
        if drugs_filter is not None:
            df = df[df["drug"].isin(drugs_filter)]
        # Optionally restrict to top-n HVG; computed elsewhere
        rows.append(df)
    return pd.concat(rows, ignore_index=True)
```

```python
# src/cellduet/rxrx3.py
"""rxrx3-core loader, OpenPhenom embedding fetch, and EFAAR-style TVN."""
import numpy as np
from sklearn.decomposition import PCA


def typical_variation_normalization(
    X: np.ndarray,
    is_control: np.ndarray,
    n_components: int = None,
) -> np.ndarray:
    """EFAAR-style TVN: PCA-fit on controls, project all wells, standardize.

    Reference: rxrx3-core paper (arXiv:2503.20158) and Recursion's
    EFAAR_benchmarking library.
    """
    n_components = n_components or X.shape[1]
    pca = PCA(n_components=n_components, whiten=True)
    pca.fit(X[is_control])
    return pca.transform(X)
```

```python
# src/cellduet/stats.py
"""Cross-modal phenotype concordance statistics for cellduet."""
from typing import Callable, Tuple

import numpy as np
from scipy.spatial.distance import squareform


def _upper_triangle(D: np.ndarray) -> np.ndarray:
    return D[np.triu_indices_from(D, k=1)]


def mantel(D1: np.ndarray, D2: np.ndarray, n_permutations: int = 999, seed: int = 0) -> Tuple[float, float]:
    """Mantel correlation between two square distance matrices.

    Returns (r, p) where r is Pearson correlation of the upper-triangle
    flattened distances, and p is a permutation p-value (or NaN if n_permutations=0).
    """
    if D1.shape != D2.shape or D1.shape[0] != D1.shape[1]:
        raise ValueError("D1 and D2 must be the same square shape")
    v1 = _upper_triangle(D1)
    v2 = _upper_triangle(D2)
    r = float(np.corrcoef(v1, v2)[0, 1])
    if n_permutations == 0:
        return r, float("nan")
    rng = np.random.default_rng(seed)
    n = D1.shape[0]
    null = np.empty(n_permutations)
    for i in range(n_permutations):
        perm = rng.permutation(n)
        v2_perm = _upper_triangle(D2[np.ix_(perm, perm)])
        null[i] = np.corrcoef(v1, v2_perm)[0, 1]
    # two-sided p-value
    p = (np.abs(null) >= abs(r)).mean()
    return r, float(p)


def rv_coefficient(D1: np.ndarray, D2: np.ndarray) -> float:
    """RV coefficient between two distance matrices, treated as kernel matrices."""
    A = -0.5 * D1**2
    B = -0.5 * D2**2
    num = float(np.trace(A @ B))
    denom = float(np.sqrt(np.trace(A @ A) * np.trace(B @ B)))
    return num / denom if denom > 0 else 0.0


def neighborhood_jaccard(D1: np.ndarray, D2: np.ndarray, k: int = 10) -> np.ndarray:
    """Per-row Jaccard overlap of the k-nearest-neighbor sets in two distance matrices."""
    n = D1.shape[0]
    nn1 = np.argsort(D1, axis=1)[:, 1 : k + 1]
    nn2 = np.argsort(D2, axis=1)[:, 1 : k + 1]
    out = np.empty(n)
    for i in range(n):
        a = set(nn1[i])
        b = set(nn2[i])
        union = len(a | b)
        out[i] = len(a & b) / union if union > 0 else 0.0
    return out


def permutation_null(
    D1: np.ndarray,
    D2: np.ndarray,
    statistic: Callable,
    n_permutations: int = 999,
    seed: int = 0,
) -> np.ndarray:
    """Null distribution of `statistic` under random row/col permutations of D2."""
    rng = np.random.default_rng(seed)
    n = D1.shape[0]
    null = np.empty(n_permutations)
    for i in range(n_permutations):
        perm = rng.permutation(n)
        D2_perm = D2[np.ix_(perm, perm)]
        result = statistic(D1, D2_perm, n_permutations=0) if statistic is mantel else statistic(D1, D2_perm)
        null[i] = result[0] if isinstance(result, tuple) else result
    return null
```
